generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

model Empresa {
  id        String   @id @default(uuid())
  nome      String
  telefone  String   @unique
  criadoEm  DateTime @default(now())
  clientes  Cliente[]
}

model Cliente {
  id        String   @id @default(uuid())
  nome      String
  telefone  String
  empresaId String
  empresa   Empresa  @relation(fields: [empresaId], references: [id])
  criadoEm  DateTime @default(now())
}
